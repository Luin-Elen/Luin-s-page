v1

手写梯度下降 · 原生态实现

手动用 GradientTape 计算梯度、用 Adam 更新参数，最贴近数学推导的一版。

import tensorflow as tf
tf.keras.backend.set_floatx("float64")

x = tf.constant([[0.,0.],[0.,1.],[1.,0.],[1.,1.]])
target = tf.constant([[0.],[1.],[1.],[0.]])

W = tf.Variable([[2.,-2.],[-2.,2.]])
b_hidden = tf.Variable([-1.,-1.])
v = tf.Variable([[3.],[3.]])
b_output = tf.Variable([-2.5])
parameters = [W, b_hidden, v, b_output]

optimizer = tf.keras.optimizers.Adam(learning_rate=0.03)
for epoch in range(1, 301):
    with tf.GradientTape() as tape:
        hidden = tf.sigmoid(tf.matmul(x, W) + b_hidden)
        logits = tf.matmul(hidden, v) + b_output

        loss = tf.reduce_mean(
            tf.nn.sigmoid_cross_entropy_with_logits(
                labels=target, logits=logits))

    gradients = tape.gradient(loss, parameters)
    optimizer.apply_gradients(zip(gradients, parameters))

hidden = tf.sigmoid(tf.matmul(x, W) + b_hidden)
prob = tf.sigmoid(tf.matmul(hidden, v) + b_output)
print(tf.cast(prob >= 0.5, tf.int32).numpy().ravel())
v2

Keras Sequential · 高层封装

改用 Sequential 与 Dense 层，把训练循环交给 model.fit，代码更简洁。

import tensorflow as tf

tf.keras.backend.set_floatx("float64")

x = tf.constant([[0.,0.], [0.,1.], [1.,0.], [1.,1.]], dtype=tf.float64)
target = tf.constant([[0.], [1.], [1.], [0.]], dtype=tf.float64)

model = tf.keras.Sequential([
    tf.keras.layers.Dense(units=2, activation='sigmoid', input_shape=(2,)),
    tf.keras.layers.Dense(units=1, activation=None)
])

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=0.03),
    loss=tf.keras.losses.BinaryCrossentropy(from_logits=True)
)

model.fit(x, target, epochs=300, verbose=0)

logits = model(x)
prob = tf.sigmoid(logits)
pred = tf.cast(prob >= 0.5, tf.int32)
print(pred.numpy().ravel())
